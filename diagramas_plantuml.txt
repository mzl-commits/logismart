@startuml diagrama_flujo

title Flujo Principal del Sistema LogiSmart

start

:Ingresa nueva caja al sistema;
:Registrar datos básicos\n(producto, peso, medida, proveedor);
:Caja en estado: **PENDIENTE**;

if (¿Procesar lote?) then (Sí)
    :Obtener configuración del carro;
    :Filtrar cajas pendientes\npor peso, volumen y paradas;
    
    partition "Clasificación Automática" {
        :Clasificar caja por:\n- Categoría\n- Fragilidad\n- Peso\n- Dimensiones;
    }
    
    partition "Optimización de Ubicación" {
        :Buscar ubicaciones libres compatibles;
        if (¿Hay ubicación disponible?) then (Sí)
            :Calcular score de compatibilidad\n(tipo estante, categoría, peso);
            :Asignar mejor ubicación;
        else (No)
            :Registrar caja sin ubicación;
            stop
        endif
    }
    
    :Cambiar estado a **EN_TRÁNSITO**;
    :Registrar en historial de movimientos;
    :Marcar ubicación como ocupada;
    
    partition "Ruta del Carro AGV" {
        :Optimizar orden de paradas (Nearest Neighbor);
        :Generar ruta A* hasta primera parada;
        :Enviar coordenadas a ESP32;
        :Estado carro: **MOVIENDO**;
    }
    
    partition "Ejecución de Ruta" {
        repeat
            :Carro avanza un paso;
            if (¿Llegó a destino?) then (Sí)
                :Estado carro: **LLEGÓ**;
                :Operador confirma entrega;
                :Cambiar estado caja a **ALMACENADA**;
                :Registrar en historial;
                if (¿Hay más paradas?) then (Sí)
                    :Calcular ruta a siguiente parada;
                    :Estado carro: **MOVIENDO**;
                else (No)
                    :Calcular ruta de regreso a base;
                    :Estado carro: **REGRESANDO**;
                endif
            else (No)
                :Continuar avanzando;
            endif
        repeat while (¿Carro en movimiento?) is (Sí)
        :Carro llegó a base;
        :Estado carro: **ESPERANDO**;
    }

else (No — Individual)
    :Procesar caja individual;
    :Clasificar y asignar ubicación;
    :Generar ruta individual;
endif

:Caja almacenada en ubicación asignada;

partition "Proceso de Despacho" {
    :Despachador selecciona cajas;
    :Asignar vehículo y destino;
    :Confirmar despacho;
    :Estado caja: **DESPACHADA**;
    :Liberar ubicación en almacén;
    :Registrar despacho en historial;
    :Crear registro en tabla Despacho;
}

stop

@enduml

title LogiSmart - Diagrama de Secuencia

actor Operador as OP
actor Despachador as DS
participant Frontend as FE
participant "API Django" as API
participant Servicios as SVC
database BD

== 1. Registro de Caja ==

OP -> FE : Llena formulario nueva caja
FE -> API : POST /api/cajas/
API -> BD : INSERT caja (estado=pendiente)
BD --> API : OK, id generado
API --> FE : 201 Created {id, estado}
FE --> OP : Notificacion de exito

== 2. Procesamiento de Lote ==

OP -> FE : Click "Procesar Lote"
FE -> API : POST /api/cajas/procesar_lote/
API -> BD : SELECT cajas pendientes
BD --> API : Lista de cajas

note over API : Filtra por peso_max,\nvolumen_max y max_paradas

loop Por cada caja seleccionada
    API -> SVC : ClasificadorCajas.clasificar(caja)
    SVC --> API : {categoria, fragilidad, peso_clase}
    API -> SVC : OptimizadorUbicaciones\n.encontrar_mejor_ubicacion()
    SVC -> BD : SELECT ubicaciones libres
    BD --> SVC : Resultados
    SVC --> API : Mejor ubicacion + score
    API -> BD : UPDATE caja a en_transito
    API -> BD : UPDATE ubicacion a ocupada
    API -> BD : INSERT historial movimiento
end

== 3. Generacion de Ruta ==

API -> SVC : RutaService.optimizar_paradas()
note right of SVC : Nearest Neighbor:\nordena por cercania
SVC --> API : Paradas ordenadas []

API -> SVC : RutaService.generar_ruta(origen, destino)
note right of SVC : BFS/A*:\ngénera lista {x,y}
SVC --> API : Ruta [{x,y}, ...]

API -> BD : UPDATE estado_carro\n(moviendo, ruta, paradas)
API -> SVC : ESP32.enviar_coordenadas(x, y)
SVC --> API : Resultado conexion
API --> FE : 200 OK {paradas, ruta}
FE --> OP : Mapa actualizado

== 4. Avance del Carro ==

OP -> FE : Click "Avanzar"
FE -> API : POST /api/carro/avanzar/
API -> BD : SELECT estado_carro
BD --> API : carro {ruta, pos}
API -> BD : UPDATE pos_x, pos_y

alt Carro llego al destino
    API -> BD : UPDATE carro = llego
    API --> FE : {estado: llego}
    FE --> OP : Boton "Confirmar entrega"

    OP -> FE : Click "Confirmar entrega"
    FE -> API : POST /api/carro/confirmar_parada/
    API -> BD : UPDATE caja a almacenada
    API -> BD : INSERT historial (en_transito a almacenada)

    alt Quedan mas paradas
        API -> SVC : generar_ruta(pos, sig_parada)
        SVC --> API : Nueva ruta
        API -> BD : UPDATE carro (moviendo)
        API -> SVC : ESP32.enviar_coordenadas()
    else Ultima parada
        API -> SVC : generar_ruta(pos, base)
        SVC --> API : Ruta de regreso
        API -> BD : UPDATE carro (regresando)
    end
else Carro en camino
    API --> FE : {estado: moviendo, pos: {x,y}}
    FE -> FE : Actualiza mapa
end

== 5. Despacho ==

DS -> FE : Selecciona cajas + vehiculo + destino
FE -> API : POST /api/cajas/{id}/confirmar_despacho/
API -> BD : UPDATE caja a despachada
API -> BD : UPDATE ubicacion a libre
API -> BD : INSERT historial (almacenada a despachada)
API -> BD : INSERT registro Despacho
API --> FE : 200 OK
FE --> DS : Despacho confirmado

@enduml
